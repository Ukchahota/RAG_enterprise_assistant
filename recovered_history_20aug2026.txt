function Get-DocumentVersion {`
    param([string]$Name)`
`
    # Examples detected:`
    # 2025-26`
    # 2025_2026`
    # 2026,2027`
    # 2026/27`
    if ($Name -match "(20\d{2})[\s,_/-]*(20\d{2}|\d{2})") {`
        return "$($matches[1])-$($matches[2])"`
    }`
`
    if ($Name -match "(20\d{2})") {`
        return $matches[1]`
    }`
`
    return ""`
}

function Find-Pdf {`
    param(`
        [Parameter(Mandatory)]`
        [string]$Pattern`
    )`
`

        Get-ChildItem ``
            -Path $rawDocumentsPath ``
            -Filter "*.pdf" ``

        Where-Object {`
            $_.BaseName -match $Pattern`
        }`
    )`
`
    if ($matches.Count -eq 0) {`
        Write-Warning "No PDF matched pattern: $Pattern"`
        return $null`
    }`
`
    if ($matches.Count -gt 1) {`
        Write-Warning "More than one PDF matched '$Pattern'. Using: $($matches[0].Name)"`
    }`
`
    return $matches[0]`
}

`
        [Parameter(Mandatory)]`

        [string]$SourceUrl,`

        Where-Object {`
            $_.file_name -ieq $Pdf.Name`

        Select-Object -First 1`
`

        $existingRecord.document_name = $DocumentName`

        $existingRecord.source = "University of Liverpool"`

        $existingRecord.used_in_rag = "Yes"`
`

            $existingRecord.notes = $Notes`
        }`
`
        Write-Host "Updated: $($Pdf.Name)"`

        $newRecord = [PSCustomObject]@{`
            document_name = $DocumentName`
            document_type = $DocumentType`
            source         = "University of Liverpool"`
            source_url     = $SourceUrl`
            file_name      = $Pdf.Name`
            date_accessed  = Get-Date -Format "dd-MM-yyyy"`
            used_in_rag    = "Yes"`
            notes          = $Notes`
        }`

`
        Write-Host "Added:   $($Pdf.Name)"`

def retrieve_with_loaded_model(`

def hit_at_k(`
    results: pd.DataFrame,`
    gold_chunk_ids: list[str],`
    k: int,`
) -> bool:`
    """`
    Return True if at least one gold chunk occurs`
    within the top-k retrieved chunks.`
    """`
    retrieved_ids = set(`
        results.head(k)["chunk_id"]`
        .astype(str)`
        .str.strip()`
        .tolist()`
    )`
`
    gold_ids = set(gold_chunk_ids)`
`
    return bool(`
        retrieved_ids.intersection(gold_ids)`
    )`
`

def reciprocal_rank(`
    results: pd.DataFrame,`
    gold_chunk_ids: list[str],`
    k: int = 5,`
) -> float:`
    """`
    Reciprocal rank of the first exact gold chunk`
    found in the top-k results.`
    """`
    gold_ids = set(gold_chunk_ids)`
`
    for _, row in results.head(k).iterrows():`
        retrieved_chunk_id = normalise(`
            row["chunk_id"]`
        )`
`
        if retrieved_chunk_id in gold_ids:`
            return 1.0 / float(`
                row["rank"]`
            )`
`
    return 0.0`
`

def precision_at_k(`
    results: pd.DataFrame,`
    gold_chunk_ids: list[str],`
    k: int,`
) -> float:`
    """`
    Exact chunk-level precision@k.`
`
    Number of retrieved gold chunks divided by k.`
    """`
    retrieved_ids = (`
        results.head(k)["chunk_id"]`
        .astype(str)`
        .str.strip()`
        .tolist()`
    )`
`
    gold_ids = set(gold_chunk_ids)`
`
    relevant_count = sum(`
        chunk_id in gold_ids`
        for chunk_id in retrieved_ids`
    )`
`
    return relevant_count / k`
`

def recall_at_k(`
    results: pd.DataFrame,`
    gold_chunk_ids: list[str],`
    k: int,`
) -> float:`
    """`
    Fraction of the question's gold chunks retrieved`
    within the top-k results.`
    """`
    gold_ids = set(gold_chunk_ids)`
`
    if not gold_ids:`
        return 0.0`
`
    retrieved_ids = set(`
        results.head(k)["chunk_id"]`
        .astype(str)`
        .str.strip()`
        .tolist()`
    )`
`
    relevant_retrieved = len(`
        retrieved_ids.intersection(`
            gold_ids`
        )`
    )`
`
    return (`
        relevant_retrieved`
        / len(gold_ids)`
    )`

def safe_join(series: pd.Series) -> str:`
    values = []`
`

import pandas as pd`
import numpy as np`
`
from sentence_transformers import SentenceTransformer`
from src.hybrid_retriever import HybridRetriever`
`
`
questions = pd.read_csv(`
    "data/evaluation/test_questions_v2_answerable.csv",`
    encoding="utf-8-sig",`
)`
`
metadata = pd.read_csv(`
    "vector_store/chunk_metadata.csv",`
    encoding="utf-8-sig",`
)`
`
questions = questions[`
    questions["question_id"].isin(`
        [`
            "V2Q098",`
            "V2Q099",`
            "V2Q100",`
        ]`
    )`
]`
`
`
embedding_model = SentenceTransformer(`
    "sentence-transformers/all-MiniLM-L6-v2"`
)`
`
retriever = HybridRetriever(`
    dense_weight=0.5,`
    bm25_weight=0.5,`
    rrf_k=60,`
)`
`
`
document_names = (`
    metadata["document_name"]`
    .astype(str)`
    .drop_duplicates()`
    .tolist()`
)`
`
document_embeddings = embedding_model.encode(`
    document_names,`
    normalize_embeddings=True,`
)`
`
`
print(`
    "question_id | gold_document | semantic_rank | best_hybrid_rank"`
)`
`
`
for _, row in questions.iterrows():`
`
    question = str(`
        row["question"]`
    )`
`
    gold_ids = [`
        x.strip()`
        for x in str(`
            row["gold_chunk_ids"]`
        ).split("|")`
    ]`
`
    question_embedding = embedding_model.encode(`
        [question],`
        normalize_embeddings=True,`
    )[0]`
`
    similarities = (`
        document_embeddings`
        @ question_embedding`
    )`
`
    semantic_order = np.argsort(`
        similarities`
    )[::-1]`
`
    semantic_rank_map = {`
        document_names[index]:`
            rank + 1`
        for rank, index`
        in enumerate(`
            semantic_order`
        )`
    }`
`
    hybrid = retriever.retrieve(`
        question,`
        top_k=50,`
        candidate_k=50,`
    )`
`
    seen_documents = set()`
`
    for gold_id in gold_ids:`
`
        gold_match = metadata[`
            metadata["chunk_id"]`
            .astype(str)`
            == gold_id`
        ]`
`
        gold_document = str(`
            gold_match.iloc[0][`
                "document_name"`
            ]`
        )`
`
        if gold_document in seen_documents:`
            continue`
`
        seen_documents.add(`
            gold_document`
        )`
`
        semantic_rank = (`
            semantic_rank_map[`
                gold_document`
            ]`
        )`
`
        hybrid_doc = hybrid[`
            hybrid["document_name"]`
            .astype(str)`
            == gold_document`
        ]`
`
        if hybrid_doc.empty:`
            best_hybrid_rank = (`
                "NOT_IN_TOP50"`
            )`
        else:`
            best_hybrid_rank = int(`
                hybrid_doc[`
                    "rank"`
                ].min()`
            )`
`
        print(`
            row["question_id"],`
            "|",`
            gold_document,`
            "|",`
            semantic_rank,`
            "|",`
            best_hybrid_rank,`
        )`
